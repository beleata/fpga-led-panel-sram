cmake_minimum_required(VERSION 3.20)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(pico_panel C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
pico_sdk_init()
add_executable(pico_panel main.cpp)
target_include_directories(pico_panel PRIVATE ${CMAKE_CURRENT_LIST_DIR}/generated)
pico_generate_pio_header(pico_panel ${CMAKE_CURRENT_LIST_DIR}/panel_wave.pio)
target_link_libraries(pico_panel pico_stdlib hardware_pio hardware_dma hardware_clocks)
pico_enable_stdio_usb(pico_panel 1)
pico_enable_stdio_uart(pico_panel 0)
pico_set_program_name(pico_panel "ICND1065L Pico static")
pico_set_program_version(pico_panel "0.1.0")
pico_add_extra_outputs(pico_panel)
